   151→   151→export const useAuthStore = create<AuthState>()((set, get) => ({
   152→   152→  currentUser: null,
   153→   153→  uid: null,
   154→   154→  userEmail: null,
   155→   155→  isLocked: false,
   156→   156→  isAuthenticated: false,
   157→   157→  isInitializing: true,
   158→   158→  needsEmailUpdate: false,
   159→   159→
   160→   160→  initAuth: () => {
   161→   161→    if (!auth) {
   162→   162→      set({ isInitializing: false });
   163→   163→      return () => {};
   164→   164→    }
   165→   165→
   166→   166→    const unsubscribe = onAuthStateChanged(auth, async (firebaseUser) => {
   167→   167→      if (firebaseUser) {
   168→   168→        const email = firebaseUser.email || "";
   169→   169→        const username = firebaseUser.displayName || email.split("@")[0];
   170→   170→
   171→   171→        // Load recovery email from Firestore (separate from auth email)
   172→   172→        let recoveryEmail: string | null = null;
   173→   173→        if (db) {
   174→   174→          try {
   175→   175→            const usernameDoc = await getDoc(doc(db, "usernames", username));
   176→   176→            if (usernameDoc.exists()) {
   177→   177→              const data = usernameDoc.data();
   178→   178→              // Read from recoveryEmail field (new), fall back to email field (old)
   179→   179→              const stored = data.recoveryEmail || data.email;
   180→   180→              if (stored && !stored.endsWith("@et.app")) {
   181→   181→                recoveryEmail = stored;
   182→   182→              }
   183→   183→            }
   184→   184→          } catch { /* ignore */ }
   185→   185→        }
   186→   186→
   187→   187→        set({
   188→   188→          currentUser: username,
   189→   189→          uid: firebaseUser.uid,
   190→   190→          userEmail: recoveryEmail,
   191→   191→          isAuthenticated: true,
   192→   192→          isInitializing: false,
   193→   193→          isLocked: false,
   194→   194→          needsEmailUpdate: !recoveryEmail,
   195→   195→        });
   196→   196→      } else {
   197→   197→        set({
   198→   198→          currentUser: null,
   199→   199→          uid: null,
   200→   200→          userEmail: null,
   201→   201→          isAuthenticated: false,
   202→   202→          isInitializing: false,
   203→   203→          isLocked: false,
   204→   204→          needsEmailUpdate: false,
   205→   205→        });
   206→   206→      }
   207→   207→    });
   208→   208→
   209→   209→    return () => unsubscribe();
   210→   210→  },
   211→   211→
   212→   212→  signup: async (username, email, password) => {
   213→   213→    if (!auth || !db) {
   214→   214→      return { success: false, error: "Firebase is not configured. See .env.local" };
   215→   215→    }
   216→   216→    const normalized = username.trim().toLowerCase();
   217→   217→    const trimmedEmail = email.trim().toLowerCase();
   218→   218→    if (normalized.length < 2) {
   219→   219→      return { success: false, error: "Username must be at least 2 characters" };
   220→   220→    }
   221→   221→    // Email is optional — validate only if provided
   222→   222→    if (trimmedEmail && (!trimmedEmail.includes("@") || !trimmedEmail.includes("."))) {
   223→   223→      return { success: false, error: "Please enter a valid email address" };
   224→   224→    }
   225→   225→    if (password.length < 6) {
   226→   226→      return { success: false, error: "Password must be at least 6 characters" };
   227→   227→    }
   228→   228→    try {
   229→   229→      // Check if username is already taken
   230→   230→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   231→   231→      if (usernameDoc.exists()) {
   232→   232→        return { success: false, error: "Username is already taken. Choose a different username." };
   233→   233→      }
   234→   234→
   235→   235→      // Use real email if provided, otherwise fall back to @et.app
   236→   236→      const fbAuthEmail = trimmedEmail || usernameToEmail(normalized);
   237→   237→
   238→   238→      // Try creating the Firebase Auth user FIRST
   239→   239→      // This handles the case where a user was deleted from Firebase Console
   240→   240→      // but the usernames doc still exists in Firestore
   241→   241→      const cred = await createUserWithEmailAndPassword(auth, fbAuthEmail, password);
   242→   242→      await updateProfile(cred.user, { displayName: normalized });
   243→   243→
   244→   244→      // Create default settings in Firestore
   245→   245→      await setDoc(doc(db, "users", cred.user.uid, "settings", "config"), {
   246→   246→        currencyCode: "BHD",
   247→   247→        monthlyBudget: 0,
   248→   248→      });
   249→   249→
   250→   250→      // Create/update username → UID mapping
   251→   251→      // authEmail = email used in Firebase Auth (for login)
   252→   252→      // recoveryEmail = email for password reset (optional, NEVER used for login)
   253→   253→      // Double-check username wasn't taken while we were creating the auth user
   254→   254→      const recheck = await getDoc(doc(db, "usernames", normalized));
   255→   255→      if (recheck.exists()) {
   256→   256→        // Username was taken — clean up the Firebase user we just created
   257→   257→        await cred.user.delete().catch(() => {});
   258→   258→        return { success: false, error: "Username is already taken. Choose a different username." };
   259→   259→      }
   260→   260→      await setDoc(doc(db, "usernames", normalized), {
   261→   261→        uid: cred.user.uid,
   262→   262→        authEmail: fbAuthEmail,
   263→   263→        recoveryEmail: trimmedEmail || null,
   264→   264→      });
   265→   265→
   266→   266→      set({
   267→   267→        currentUser: normalized,
   268→   268→        uid: cred.user.uid,
   269→   269→        userEmail: trimmedEmail || null,
   270→   270→        isAuthenticated: true,
   271→   271→        isLocked: false,
   272→   272→        needsEmailUpdate: !trimmedEmail,
   273→   273→      });
   274→   274→      return { success: true };
   275→   275→    } catch (err: unknown) {
   276→   276→      const code = (err as { code?: string })?.code || "";
   277→   277→      if (code === "auth/email-already-in-use") {
   278→   278→        return { success: false, error: "This email or username is already registered" };
   279→   279→      }
   280→   280→      if (code === "auth/weak-password") {
   281→   281→        return { success: false, error: "Password is too weak (min 6 chars)" };
   282→   282→      }
   283→   283→      if (code === "auth/invalid-email") {
   284→   284→        return { success: false, error: "Invalid email address" };
   285→   285→      }
   286→   286→      return { success: false, error: "Signup failed. Please try again." };
   287→   287→    }
   288→   288→  },
   289→   289→
   290→   290→  login: async (username, password) => {
   291→   291→    if (!auth || !db) {
   292→   292→      return { success: false, error: "Firebase is not configured. See .env.local" };
   293→   293→    }
   294→   294→    const normalized = username.trim().toLowerCase();
   295→   295→    try {
   296→   296→      // Look up the username doc to check if user exists
   297→   297→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   298→   298→      if (!usernameDoc.exists()) {
   299→   299→        return { success: false, error: "User not found. Please sign up first." };
   300→   300→      }
   301→   301→
   302→   302→      // Use authEmail for login (the email used during Firebase Auth signup)
   303→   303→      // NEVER use recoveryEmail for login — it’s only for password reset
   304→   304→      const data = usernameDoc.data();
   305→   305→      let loginEmail = data.authEmail || usernameToEmail(normalized);
   306→   306→
   307→   307→      // Auto-migrate old docs: if authEmail field doesn’t exist, set it now
   308→   308→      if (!data.authEmail) {
   309→   309→        const correctAuthEmail = usernameToEmail(normalized);
   310→   310→        updateDoc(doc(db, "usernames", normalized), { authEmail: correctAuthEmail }).catch(() => {});
   311→   311→        loginEmail = correctAuthEmail;
   312→   312→      }
   313→   313→
   314→   314→      await signInWithEmailAndPassword(auth, loginEmail, password);
   315→   315→      return { success: true };
   316→   316→    } catch (err: unknown) {
   317→   317→      const code = (err as { code?: string })?.code || "";
   318→   318→      // Username exists (we checked above), so it must be wrong password
   319→   319→      if (code === "auth/invalid-credential" || code === "auth/wrong-password") {
   320→   320→        return { success: false, error: "Incorrect password" };
   321→   321→      }
   322→   322→      if (code === "auth/user-not-found") {
   323→   323→        // Orphaned Firestore doc — Auth user was deleted (e.g. from Firebase Console)
   324→   324→        // Clean up ALL remaining Firestore data for this user
   325→   325→        try {
   326→   326→          const data = (await getDoc(doc(db, "usernames", normalized))).data();
   327→   327→          const uid = data?.uid;
   328→   328→          if (uid) {
   329→   329→            // Delete all expenses
   330→   330→            const expensesSnapshot = await getDocs(collection(db, "users", uid, "expenses"));
   331→   331→            if (!expensesSnapshot.empty) {
   332→   332→              const batch = writeBatch(db);
   333→   333→              expensesSnapshot.docs.forEach((d) => batch.delete(d.ref));
   334→   334→              await batch.commit();
   335→   335→            }
   336→   336→            // Delete settings
   337→   337→            await deleteDoc(doc(db, "users", uid, "settings", "config")).catch(() => {});
   338→   338→          }
   339→   339→        } catch { /* best effort cleanup */ }
   340→   340→        // Finally delete the usernames doc
   341→   341→        await deleteDoc(doc(db, "usernames", normalized)).catch(() => {});
   342→   342→        return { success: false, error: "User not found. Please sign up first." };
   343→   343→      }
   344→   344→      return { success: false, error: "Login failed. Please try again." };
   345→   345→    }
   346→   346→  },
   347→   347→
   348→   348→  logout: async () => {
   349→   349→    if (!auth) return;
   350→   350→    try {
   351→   351→      await signOut(auth);
   352→   352→    } catch {
   353→   353→      // ignore sign out errors
   354→   354→    }
   355→   355→  },
   356→   356→
   357→   357→  lock: () => {
   358→   358→    set({ isLocked: true });
   359→   359→  },
   360→   360→
   361→   361→  unlock: async (password) => {
   362→   362→    if (!auth || !auth.currentUser) return false;
   363→   363→    try {
   364→   364→      // Use the actual email from Firebase Auth (works for both old and new users)
   365→   365→      const email = auth.currentUser.email!;
   366→   366→      const credential = EmailAuthProvider.credential(email, password);
   367→   367→      await reauthenticateWithCredential(auth.currentUser, credential);
   368→   368→      set({ isLocked: false });
   369→   369→      return true;
   370→   370→    } catch {
   371→   371→      return false;
   372→   372→    }
   373→   373→  },
   374→   374→
   375→   375→  deleteAccount: async (password) => {
   376→   376→    if (!auth || !db || !auth.currentUser) {
   377→   377→      return { success: false, error: "Not authenticated" };
   378→   378→    }
   379→   379→    const { uid, currentUser } = get();
   380→   380→    if (!uid) return { success: false, error: "Not authenticated" };
   381→   381→    try {
   382→   382→      // Re-authenticate (use actual email from Firebase Auth)
   383→   383→      const email = auth.currentUser.email!;
   384→   384→      const credential = EmailAuthProvider.credential(email, password);
   385→   385→      await reauthenticateWithCredential(auth.currentUser, credential);
   386→   386→
   387→   387→      // Delete all expenses from Firestore
   388→   388→      const expensesRef = collection(db, "users", uid, "expenses");
   389→   389→      const snapshot = await getDocs(expensesRef);
   390→   390→      if (!snapshot.empty) {
   391→   391→        const batch = writeBatch(db);
   392→   392→        snapshot.docs.forEach((d) => batch.delete(d.ref));
   393→   393→        await batch.commit();
   394→   394→      }
   395→   395→
   396→   396→      // Delete settings document
   397→   397→      try {
   398→   398→        await deleteDoc(doc(db, "users", uid, "settings", "config"));
   399→   399→      } catch {
   400→   400→        // settings doc might not exist
   401→   401→      }
   402→   402→
   403→   403→      // Delete username mapping
   404→   404→      if (currentUser) {
   405→   405→        try {
   406→   406→          await deleteDoc(doc(db, "usernames", currentUser));
   407→   407→        } catch {
   408→   408→          // ignore
   409→   409→        }
   410→   410→      }
   411→   411→
   412→   412→      // Delete the Firebase Auth user
   413→   413→      await firebaseDeleteUser(auth.currentUser);
   414→   414→      return { success: true };
   415→   415→    } catch (err: unknown) {
   416→   416→      const code = (err as { code?: string })?.code || "";
   417→   417→      if (code === "auth/wrong-password" || code === "auth/invalid-credential") {
   418→   418→        return { success: false, error: "Incorrect password" };
   419→   419→      }
   420→   420→      return { success: false, error: "Failed to delete account. Please try again." };
   421→   421→    }
   422→   422→  },
   423→   423→
   424→   424→  forgotPassword: async (username) => {
   425→   425→    if (!auth || !db) {
   426→   426→      return { success: false, error: "Firebase not configured" };
   427→   427→    }
   428→   428→    const normalized = username.trim().toLowerCase();
   429→   429→    try {
   430→   430→      // Look up the user’s recovery email from Firestore
   431→   431→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   432→   432→      if (!usernameDoc.exists()) {
   433→   433→        return { success: false, error: "Username not found" };
   434→   434→      }
   435→   435→      const data = usernameDoc.data();
   436→   436→      // Use recoveryEmail field (new), fall back to email field (old docs)
   437→   437→      const recoveryEmail = data.recoveryEmail || data.email;
   438→   438→
   439→   439→      // Check if the user has a real recovery email
   440→   440→      if (!recoveryEmail || recoveryEmail.endsWith("@et.app")) {
   441→   441→        return {
   442→   442→          success: false,
   443→   443→          error: "This account was created without a recovery email. Please create a new account with your email to enable password reset.",
   444→   444→        };
   445→   445→      }
   446→   446→
   447→   447→      // Send Firebase password reset email
   448→   448→      await sendPasswordResetEmail(auth, recoveryEmail, {
   449→   449→        url: "https://expense-tracker-five-alpha-69.vercel.app/",
   450→   450→      });
   451→   451→      return { success: true, email: recoveryEmail };
   452→   452→    } catch {
   453→   453→      return { success: false, error: "Failed to send reset email. Please try again." };
   454→   454→    }
   455→   455→  },
   456→   456→
   457→   457→  updateUserEmail: async (newEmail, password) => {
   458→   458→    if (!auth?.currentUser) {
   459→   459→      return { success: false, error: "Not authenticated" };
   460→   460→    }
   461→   461→    const { currentUser } = get();
   462→   462→    if (!currentUser || !db) {
   463→   463→      return { success: false, error: "Not authenticated" };
   464→   464→    }
   465→   465→    const trimmedEmail = newEmail.trim().toLowerCase();
   466→   466→    if (!trimmedEmail || !trimmedEmail.includes("@") || !trimmedEmail.includes(".")) {
   467→   467→      return { success: false, error: "Please enter a valid email address" };
   468→   468→    }
   469→   469→    if (!password || password.length < 6) {
   470→   470→      return { success: false, error: "Please enter your password to confirm" };
   471→   471→    }
   472→   472→
   473→   473→    // Check if this email is already used by another account

   509→   509→          }
   510→   510→        }

   545→   545→    return { success: true };
   546→   546→  },
