    79→export function formatCurrency(amount: number, currencyCode: string): string {
    80→  const currency = getCurrency(currencyCode);
    81→  const formatted = amount.toLocaleString("en-US", {
    82→    minimumFractionDigits: currency.decimals,

   151→export const useAuthStore = create<AuthState>()((set, get) => ({
   152→  currentUser: null,
   153→  uid: null,
   154→  userEmail: null,
   155→  isLocked: false,
   156→  isAuthenticated: false,
   157→  isInitializing: true,
   158→  needsEmailUpdate: false,
   159→
   160→  initAuth: () => {
   161→    if (!auth) {
   162→      set({ isInitializing: false });
   163→      return () => {};
   164→    }
   165→
   166→    const unsubscribe = onAuthStateChanged(auth, async (firebaseUser) => {
   167→      if (firebaseUser) {
   168→        const email = firebaseUser.email || "";
   169→        const username = firebaseUser.displayName || email.split("@")[0];
   170→
   171→        // Load recovery email from Firestore (separate from auth email)
   172→        let recoveryEmail: string | null = null;
   173→        if (db) {
   174→          try {
   175→            const usernameDoc = await getDoc(doc(db, "usernames", username));
   176→            if (usernameDoc.exists()) {
   177→              const data = usernameDoc.data();
   178→              // Read from recoveryEmail field (new), fall back to email field (old)
   179→              const stored = data.recoveryEmail || data.email;
   180→              if (stored && !stored.endsWith("@et.app")) {
   181→                recoveryEmail = stored;
   182→              }
   183→            }
   184→          } catch { /* ignore */ }
   185→        }
   186→
   187→        set({
   188→          currentUser: username,
   189→          uid: firebaseUser.uid,
   190→          userEmail: recoveryEmail,
   191→          isAuthenticated: true,
   192→          isInitializing: false,
   193→          isLocked: false,
   194→          needsEmailUpdate: !recoveryEmail,
   195→        });
   196→      } else {
   197→        set({
   198→          currentUser: null,
   199→          uid: null,
   200→          userEmail: null,
   201→          isAuthenticated: false,
   202→          isInitializing: false,
   203→          isLocked: false,
   204→          needsEmailUpdate: false,
   205→        });
   206→      }
   207→    });
   208→
   209→    return () => unsubscribe();
   210→  },
   211→
   212→  signup: async (username, email, password) => {
   213→    if (!auth || !db) {
   214→      return { success: false, error: "Firebase is not configured. See .env.local" };
   215→    }
   216→    const normalized = username.trim().toLowerCase();
   217→    const trimmedEmail = email.trim().toLowerCase();
   218→    if (normalized.length < 2) {
   219→      return { success: false, error: "Username must be at least 2 characters" };
   220→    }
   221→    // Email is optional — validate only if provided
   222→    if (trimmedEmail && (!trimmedEmail.includes("@") || !trimmedEmail.includes("."))) {
   223→      return { success: false, error: "Please enter a valid email address" };
   224→    }
   225→    if (password.length < 6) {
   226→      return { success: false, error: "Password must be at least 6 characters" };
   227→    }
   228→    try {
   229→      // Check if username is already taken
   230→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   231→      if (usernameDoc.exists()) {
   232→        return { success: false, error: "Username is already taken. Choose a different username." };
   233→      }
   234→
   235→      // Use real email if provided, otherwise fall back to @et.app
   236→      const fbAuthEmail = trimmedEmail || usernameToEmail(normalized);
   237→
   238→      // Try creating the Firebase Auth user FIRST
   239→      // This handles the case where a user was deleted from Firebase Console
   240→      // but the usernames doc still exists in Firestore
   241→      const cred = await createUserWithEmailAndPassword(auth, fbAuthEmail, password);
   242→      await updateProfile(cred.user, { displayName: normalized });
   243→
   244→      // Create default settings in Firestore
   245→      await setDoc(doc(db, "users", cred.user.uid, "settings", "config"), {
   246→        currencyCode: "BHD",
   247→        monthlyBudget: 0,
   248→      });
   249→
   250→      // Create/update username → UID mapping
   251→      // authEmail = email used in Firebase Auth (for login)
   252→      // recoveryEmail = email for password reset (optional, NEVER used for login)
   253→      // Double-check username wasn't taken while we were creating the auth user
   254→      const recheck = await getDoc(doc(db, "usernames", normalized));
   255→      if (recheck.exists()) {
   256→        // Username was taken — clean up the Firebase user we just created
   257→        await cred.user.delete().catch(() => {});
   258→        return { success: false, error: "Username is already taken. Choose a different username." };
   259→      }
   260→      await setDoc(doc(db, "usernames", normalized), {
   261→        uid: cred.user.uid,
   262→        authEmail: fbAuthEmail,
   263→        recoveryEmail: trimmedEmail || null,
   264→      });
   265→
   266→      set({
   267→        currentUser: normalized,
   268→        uid: cred.user.uid,
   269→        userEmail: trimmedEmail || null,
   270→        isAuthenticated: true,
   271→        isLocked: false,
   272→        needsEmailUpdate: !trimmedEmail,
   273→      });
   274→      return { success: true };
   275→    } catch (err: unknown) {
   276→      const code = (err as { code?: string })?.code || "";
   277→      if (code === "auth/email-already-in-use") {
   278→        return { success: false, error: "This email or username is already registered" };
   279→      }
   280→      if (code === "auth/weak-password") {
   281→        return { success: false, error: "Password is too weak (min 6 chars)" };
   282→      }
   283→      if (code === "auth/invalid-email") {
   284→        return { success: false, error: "Invalid email address" };
   285→      }
   286→      return { success: false, error: "Signup failed. Please try again." };
   287→    }
   288→  },
   289→
   290→  login: async (username, password) => {
   291→    if (!auth || !db) {
   292→      return { success: false, error: "Firebase is not configured. See .env.local" };
   293→    }
   294→    const normalized = username.trim().toLowerCase();
   295→    try {
   296→      // Look up the username doc to check if user exists
   297→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   298→      if (!usernameDoc.exists()) {
   299→        return { success: false, error: "User not found. Please sign up first." };
   300→      }
   301→
   302→      // Use authEmail for login (the email used during Firebase Auth signup)
   303→      // NEVER use recoveryEmail for login — it’s only for password reset
   304→      const data = usernameDoc.data();
   305→      let loginEmail = data.authEmail || usernameToEmail(normalized);
   306→
   307→      // Auto-migrate old docs: if authEmail field doesn’t exist, set it now
   308→      if (!data.authEmail) {
   309→        const correctAuthEmail = usernameToEmail(normalized);
   310→        updateDoc(doc(db, "usernames", normalized), { authEmail: correctAuthEmail }).catch(() => {});
   311→        loginEmail = correctAuthEmail;
   312→      }
   313→
   314→      await signInWithEmailAndPassword(auth, loginEmail, password);
   315→      return { success: true };
   316→    } catch (err: unknown) {
   317→      const code = (err as { code?: string })?.code || "";
   318→      // Username exists (we checked above), so it must be wrong password
   319→      if (code === "auth/invalid-credential" || code === "auth/wrong-password") {
   320→        return { success: false, error: "Incorrect password" };
   321→      }
   322→      if (code === "auth/user-not-found") {
   323→        // Orphaned Firestore doc — Auth user was deleted (e.g. from Firebase Console)
   324→        // Clean up ALL remaining Firestore data for this user
   325→        try {
   326→          const data = (await getDoc(doc(db, "usernames", normalized))).data();
   327→          const uid = data?.uid;
   328→          if (uid) {
   329→            // Delete all expenses
   330→            const expensesSnapshot = await getDocs(collection(db, "users", uid, "expenses"));
   331→            if (!expensesSnapshot.empty) {
   332→              const batch = writeBatch(db);
   333→              expensesSnapshot.docs.forEach((d) => batch.delete(d.ref));
   334→              await batch.commit();
   335→            }
   336→            // Delete settings
   337→            await deleteDoc(doc(db, "users", uid, "settings", "config")).catch(() => {});
   338→          }
   339→        } catch { /* best effort cleanup */ }
   340→        // Finally delete the usernames doc
   341→        await deleteDoc(doc(db, "usernames", normalized)).catch(() => {});
   342→        return { success: false, error: "User not found. Please sign up first." };
   343→      }
   344→      return { success: false, error: "Login failed. Please try again." };
   345→    }
   346→  },
   347→
   348→  logout: async () => {
   349→    if (!auth) return;
   350→    try {
   351→      await signOut(auth);
   352→    } catch {
   353→      // ignore sign out errors
   354→    }
   355→  },
   356→
   357→  lock: () => {
   358→    set({ isLocked: true });
   359→  },
   360→
   361→  unlock: async (password) => {
   362→    if (!auth || !auth.currentUser) return false;
   363→    try {
   364→      // Use the actual email from Firebase Auth (works for both old and new users)
   365→      const email = auth.currentUser.email!;
   366→      const credential = EmailAuthProvider.credential(email, password);
   367→      await reauthenticateWithCredential(auth.currentUser, credential);
   368→      set({ isLocked: false });
   369→      return true;
   370→    } catch {
   371→      return false;
   372→    }
   373→  },
   374→
   375→  deleteAccount: async (password) => {
   376→    if (!auth || !db || !auth.currentUser) {
   377→      return { success: false, error: "Not authenticated" };
   378→    }
   379→    const { uid, currentUser } = get();
   380→    if (!uid) return { success: false, error: "Not authenticated" };
   381→    try {
   382→      // Re-authenticate (use actual email from Firebase Auth)
   383→      const email = auth.currentUser.email!;
   384→      const credential = EmailAuthProvider.credential(email, password);
   385→      await reauthenticateWithCredential(auth.currentUser, credential);
   386→
   387→      // Delete all expenses from Firestore
   388→      const expensesRef = collection(db, "users", uid, "expenses");
   389→      const snapshot = await getDocs(expensesRef);
   390→      if (!snapshot.empty) {
   391→        const batch = writeBatch(db);
   392→        snapshot.docs.forEach((d) => batch.delete(d.ref));
   393→        await batch.commit();
   394→      }
   395→
   396→      // Delete settings document
   397→      try {
   398→        await deleteDoc(doc(db, "users", uid, "settings", "config"));
   399→      } catch {
   400→        // settings doc might not exist
   401→      }
   402→
   403→      // Delete username mapping
   404→      if (currentUser) {
   405→        try {
   406→          await deleteDoc(doc(db, "usernames", currentUser));
   407→        } catch {
   408→          // ignore
   409→        }
   410→      }
   411→
   412→      // Delete the Firebase Auth user
   413→      await firebaseDeleteUser(auth.currentUser);
   414→      return { success: true };
   415→    } catch (err: unknown) {
   416→      const code = (err as { code?: string })?.code || "";
   417→      if (code === "auth/wrong-password" || code === "auth/invalid-credential") {
   418→        return { success: false, error: "Incorrect password" };
   419→      }
   420→      return { success: false, error: "Failed to delete account. Please try again." };
   421→    }
   422→  },
   423→
   424→  forgotPassword: async (username) => {
   425→    if (!auth || !db) {
   426→      return { success: false, error: "Firebase not configured" };
   427→    }
   428→    const normalized = username.trim().toLowerCase();
   429→    try {
   430→      // Look up the user’s recovery email from Firestore
   431→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   432→      if (!usernameDoc.exists()) {
   433→        return { success: false, error: "Username not found" };
   434→      }
   435→      const data = usernameDoc.data();
   436→      // Use recoveryEmail field (new), fall back to email field (old docs)
   437→      const recoveryEmail = data.recoveryEmail || data.email;
   438→
   439→      // Check if the user has a real recovery email
   440→      if (!recoveryEmail || recoveryEmail.endsWith("@et.app")) {
   441→        return {
   442→          success: false,
   443→          error: "This account was created without a recovery email. Please create a new account with your email to enable password reset.",
   444→        };
   445→      }
   446→
   447→      // Send Firebase password reset email
   448→      await sendPasswordResetEmail(auth, recoveryEmail, {
   449→        url: "https://expense-tracker-five-alpha-69.vercel.app/",
   450→      });
   451→      return { success: true, email: recoveryEmail };
   452→    } catch {
   453→      return { success: false, error: "Failed to send reset email. Please try again." };
   454→    }
   455→  },
   456→
   457→  updateUserEmail: async (newEmail, password) => {
   458→    if (!auth?.currentUser) {
   459→      return { success: false, error: "Not authenticated" };
   460→    }
   461→    const { currentUser } = get();
   462→    if (!currentUser || !db) {
   463→      return { success: false, error: "Not authenticated" };
   464→    }
   465→    const trimmedEmail = newEmail.trim().toLowerCase();
   466→    if (!trimmedEmail || !trimmedEmail.includes("@") || !trimmedEmail.includes(".")) {
   467→      return { success: false, error: "Please enter a valid email address" };
   468→    }
   469→    if (!password || password.length < 6) {
   470→      return { success: false, error: "Please enter your password to confirm" };
   471→    }
   472→
   473→    // Check if this email is already used by another account
   474→    try {
   475→      const q = query(
   476→        collection(db, "usernames"),
   477→        where("recoveryEmail", "==", trimmedEmail)
   478→      );
   479→      const snapshot = await getDocs(q);
   480→      for (const d of snapshot.docs) {
   481→        if (d.id !== currentUser) {
   482→          return { success: false, error: "This email is already used by another account" };
   483→        }
   484→      }
   485→      // Also check old field name for backward compat
   486→      const q2 = query(
   487→        collection(db, "usernames"),
   488→        where("email", "==", trimmedEmail)
   489→      );
   490→      const snapshot2 = await getDocs(q2);
   491→      for (const d of snapshot2.docs) {
   492→        if (d.id !== currentUser) {
   493→          // Only flag if this doc doesn't already have recoveryEmail set
   494→          if (!d.data().recoveryEmail) {
   495→            return { success: false, error: "This email is already used by another account" };
   496→          }
   497→        }
   498→      }
   499→    } catch (err) {
   500→      // If query fails, try fallback: fetch all docs and check manually
   501→      try {
   502→        const allDocs = await getDocs(collection(db, "usernames"));
   503→        for (const d of allDocs.docs) {
   504→          if (d.id === currentUser) continue;
   505→          const data = d.data();
   506→          const em = data.recoveryEmail || data.email;
   507→          if (em && em.toLowerCase() === trimmedEmail) {
   508→            return { success: false, error: "This email is already used by another account" };
   509→          }
   510→        }
   511→      } catch {
   512→        // If even the fallback fails, allow the save (security rules may block reads)
   513→      }
   514→    }
   515→
   516→    // Step 1: Re-authenticate then update Firebase Auth email FIRST
   517→    try {
   518→      const currentAuthEmail = auth.currentUser!.email!;
   519→      const credential = EmailAuthProvider.credential(currentAuthEmail, password);
   520→      await reauthenticateWithCredential(auth.currentUser!, credential);
   521→      if (auth.currentUser!.email !== trimmedEmail) {
   522→        await firebaseUpdateEmail(auth.currentUser!, trimmedEmail);
   523→      }
   524→    } catch (authErr: unknown) {
   525→      const code = (authErr as { code?: string })?.code || "";
   526→      if (code === "auth/wrong-password" || code === "auth/invalid-credential") {
   527→        return { success: false, error: "Incorrect password. Please try again." };
   528→      }
   529→      // If Firebase Auth email update fails for any reason, don't update Firestore either
   530→      const msg = code === "auth/email-already-in-use"
   531→        ? "This email is already used by another Firebase account"
   532→        : "Failed to update email in Firebase. Please try again.";
   533→      return { success: false, error: msg };
   534→    }
   535→
   536→    // Step 2: Only after Firebase Auth succeeds, update Firestore
   537→    try {
   538→      await updateDoc(doc(db, "usernames", currentUser), {
   539→        recoveryEmail: trimmedEmail,
   540→        email: trimmedEmail,
   541→        authEmail: trimmedEmail,
   542→      });
   543→    } catch {
   544→      // Firestore update failed but Firebase Auth is already updated — acceptable
   545→    }
   546→
   547→    // Update local state
   548→    set({ userEmail: trimmedEmail, needsEmailUpdate: false });
   549→    return { success: true };
   550→  },
   551→}));

   641→// SETTINGS STORE — backed by Cloud Firestore
   642→// ============================================================
   643→
   644→interface SettingsState {
   645→  currencyCode: string;
   646→  monthlyBudget: number; // legacy single budget (kept for migration)
   647→  monthlyBudgets: Record<string, number>; // per-month budgets: "YYYY-MM" -> amount
   648→  isLoading: boolean;
   649→  _unsubscribe: Unsubscribe | null;
   650→
   651→  subscribeToSettings: (uid: string) => void;
   652→  unsubscribeFromSettings: () => void;
   653→  saveSettings: (currencyCode: string) => Promise<void>;
   654→  saveBudgetForMonth: (monthKey: string, amount: number) => Promise<void>;
   655→  getBudgetForMonth: (monthKey: string) => number;
   656→}
   657→
   658→export const useSettingsStore = create<SettingsState>()((set, get) => ({
   659→  currencyCode: "BHD",
   660→  monthlyBudget: 0,
   661→  monthlyBudgets: {},
   662→  isLoading: false,
   663→  _unsubscribe: null,
   664→
   665→  subscribeToSettings: (uid: string) => {
   666→    const prev = get()._unsubscribe;
   667→    if (prev) prev();
   668→
   669→    if (!db) return;
   670→
   671→    set({ isLoading: true });
   672→    const docRef = doc(db, "users", uid, "settings", "config");
   673→    const unsub = onSnapshot(
   674→      docRef,
   675→      (snapshot) => {
   676→        if (snapshot.exists()) {
   677→          const data = snapshot.data();
   678→          const budgets = data.monthlyBudgets || {};
   679→          const legacyBudget = data.monthlyBudget || 0;
   680→
   681→          // Migrate old single monthlyBudget to current month if no per-month budgets exist
   682→          if (legacyBudget > 0 && Object.keys(budgets).length === 0) {
   683→            const currentMonth = `${new Date().getFullYear()}-${String(new Date().getMonth() + 1).padStart(2, "0")}`;
   684→            budgets[currentMonth] = legacyBudget;
   685→            // Write migrated data back to Firestore
   686→            setDoc(docRef, { monthlyBudgets: budgets, monthlyBudget: 0 }, { merge: true }).catch(() => {});
   687→          }
   688→
   689→          set({
   690→            currencyCode: data.currencyCode || "BHD",
   691→            monthlyBudget: legacyBudget,
   692→            monthlyBudgets: budgets,
   693→            isLoading: false,
   694→          });
   695→        } else {
   696→          set({ isLoading: false });
   697→        }
   698→      },
   699→      () => {
   700→        set({ isLoading: false });
   701→      }
   702→    );
   703→
   704→    set({ _unsubscribe: unsub });
   705→  },
   706→
   707→  unsubscribeFromSettings: () => {
   708→    const prev = get()._unsubscribe;
   709→    if (prev) {
   710→      prev();
   711→      set({ _unsubscribe: null });
   712→    }
   713→  },
   714→
   715→  saveSettings: async (currencyCode) => {
   716→    const { uid } = useAuthStore.getState();
   717→    if (!db || !uid) return;
   718→    const docRef = doc(db, "users", uid, "settings", "config");
   719→    await setDoc(docRef, { currencyCode }, { merge: true });
   720→  },
   721→
   722→  saveBudgetForMonth: async (monthKey, amount) => {
   723→    const { uid } = useAuthStore.getState();
   724→    if (!db || !uid) return;
   725→    const docRef = doc(db, "users", uid, "settings", "config");
   726→
   727→    // Optimistic update — update the store immediately so UI reflects instantly
   728→    const currentBudgets = get().monthlyBudgets;
   729→    const optimistic = { ...currentBudgets };
   730→    if (amount > 0) {
   731→      optimistic[monthKey] = amount;
   732→    } else {
   733→      delete optimistic[monthKey];
   734→    }
   735→    set({ monthlyBudgets: optimistic });
   736→
   737→    // Persist to Firestore using dot notation (only updates the specific month key)
   738→    try {
   739→      if (amount > 0) {
   740→        await updateDoc(docRef, { ["monthlyBudgets." + monthKey]: amount });
   741→      } else {
   742→        await updateDoc(docRef, { ["monthlyBudgets." + monthKey]: firebaseDeleteField() });
   743→      }
   744→    } catch {
   745→      // onSnapshot will correct the store if the write fails
   746→    }
   747→  },
   748→
   749→  getBudgetForMonth: (monthKey) => {
   750→    const { monthlyBudgets } = get();
   751→    return monthlyBudgets[monthKey] || 0;
   752→  },
   753→}));
