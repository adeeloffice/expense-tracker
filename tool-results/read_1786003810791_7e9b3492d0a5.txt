    80→export function formatCurrency(amount: number, currencyCode: string): string {
    81→  const currency = getCurrency(currencyCode);
    82→  const formatted = amount.toLocaleString("en-US", {
    83→    minimumFractionDigits: currency.decimals,
    84→    maximumFractionDigits: currency.decimals,
    85→  });
    86→  return `${currency.symbol} ${formatted}`;

   152→export const useAuthStore = create<AuthState>()((set, get) => ({
   153→  currentUser: null,
   154→  uid: null,
   155→  userEmail: null,
   156→  isLocked: false,
   157→  isAuthenticated: false,
   158→  isInitializing: true,

   161→  initAuth: () => {
   162→    if (!auth) {
   163→      set({ isInitializing: false });
   164→      return () => {};
   165→    }
   166→
   167→    const unsubscribe = onAuthStateChanged(auth, async (firebaseUser) => {
   168→      if (firebaseUser) {
   169→        const email = firebaseUser.email || "";
   170→        const username = firebaseUser.displayName || email.split("@")[0];
   171→
   172→        // Load recovery email from Firestore (separate from auth email)
   173→        let recoveryEmail: string | null = null;
   174→        let storedAuthEmail: string | null = null;
   175→        if (db) {
   176→          try {
   177→            const usernameDoc = await getDoc(doc(db, "usernames", username));
   178→            if (usernameDoc.exists()) {
   179→              const data = usernameDoc.data();
   180→              // Read from recoveryEmail field (new), fall back to email field (old)
   181→              const stored = data.recoveryEmail || data.email;
   182→              if (stored && !stored.endsWith("@et.app")) {
   183→                recoveryEmail = stored;
   184→              }
   185→              storedAuthEmail = data.authEmail || null;
   186→
   187→              // Sync authEmail in Firestore if Firebase Auth email changed
   188→              // (happens after user clicks verifyBeforeUpdateEmail link)
   189→              if (email && email !== storedAuthEmail && !email.endsWith("@et.app")) {
   190→                updateDoc(doc(db, "usernames", username), { authEmail: email }).catch(() => {});
   191→              }
   192→            }
   193→          } catch { /* Firestore read failed, use fallback below */ }
   194→        }
   195→
   196→        // Fallback: if Firestore didn't have a recovery email, use the Firebase Auth email
   197→        // (when user signed up with a real email, it's stored in both places)
   198→        if (!recoveryEmail && email && !email.endsWith("@et.app")) {
   199→          recoveryEmail = email;
   200→        }
   201→
   202→        set({
   203→          currentUser: username,
   204→          uid: firebaseUser.uid,
   205→          userEmail: recoveryEmail,
   206→          isAuthenticated: true,
   207→          isInitializing: false,
   208→          isLocked: false,
   209→          needsEmailUpdate: !recoveryEmail,
   210→        });
   211→      } else {
   212→        set({
   213→          currentUser: null,
   214→          uid: null,
   215→          userEmail: null,
   216→          isAuthenticated: false,
   217→          isInitializing: false,
   218→          isLocked: false,
   219→          needsEmailUpdate: false,
   220→        });
   221→      }
   222→    });
   223→
   224→    return () => unsubscribe();
   225→  },
   226→
   227→  signup: async (username, email, password) => {
   228→    if (!auth || !db) {
   229→      return { success: false, error: "Firebase is not configured. See .env.local" };
   230→    }
   231→    const normalized = username.trim().toLowerCase();
   232→    const trimmedEmail = email.trim().toLowerCase();
   233→    if (normalized.length < 2) {
   234→      return { success: false, error: "Username must be at least 2 characters" };
   235→    }
   236→    // Email is optional — validate only if provided
   237→    if (trimmedEmail && (!trimmedEmail.includes("@") || !trimmedEmail.includes("."))) {
   238→      return { success: false, error: "Please enter a valid email address" };
   239→    }
   240→    if (password.length < 6) {
   241→      return { success: false, error: "Password must be at least 6 characters" };
   242→    }
   243→    try {
   244→      // Check if username is already taken
   245→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   246→      if (usernameDoc.exists()) {
   247→        return { success: false, error: "Username is already taken. Choose a different username." };
   248→      }
   249→
   250→      // Use real email if provided, otherwise fall back to @et.app
   251→      const fbAuthEmail = trimmedEmail || usernameToEmail(normalized);
   252→
   253→      // Try creating the Firebase Auth user FIRST
   254→      // This handles the case where a user was deleted from Firebase Console
   255→      // but the usernames doc still exists in Firestore
   256→      const cred = await createUserWithEmailAndPassword(auth, fbAuthEmail, password);
   257→      await updateProfile(cred.user, { displayName: normalized });
   258→
   259→      // Create default settings in Firestore
   260→      await setDoc(doc(db, "users", cred.user.uid, "settings", "config"), {
   261→        currencyCode: "BHD",
   262→        monthlyBudget: 0,
   263→      });
   264→
   265→      // Create/update username → UID mapping
   266→      // authEmail = email used in Firebase Auth (for login)
   267→      // recoveryEmail = email for password reset (optional, NEVER used for login)
   268→      // Double-check username wasn't taken while we were creating the auth user
   269→      const recheck = await getDoc(doc(db, "usernames", normalized));
   270→      if (recheck.exists()) {
   271→        // Username was taken — clean up the Firebase user we just created
   272→        await cred.user.delete().catch(() => {});
   273→        return { success: false, error: "Username is already taken. Choose a different username." };
   274→      }
   275→      await setDoc(doc(db, "usernames", normalized), {
   276→        uid: cred.user.uid,
   277→        authEmail: fbAuthEmail,
   278→        recoveryEmail: trimmedEmail || null,
   279→      });
   280→
   281→      set({
   282→        currentUser: normalized,
   283→        uid: cred.user.uid,
   284→        userEmail: trimmedEmail || null,
   285→        isAuthenticated: true,
   286→        isLocked: false,
   287→        needsEmailUpdate: !trimmedEmail,
   288→      });
   289→      return { success: true };
   290→    } catch (err: unknown) {
   291→      const code = (err as { code?: string })?.code || "";
   292→      if (code === "auth/email-already-in-use") {
   293→        return { success: false, error: "This email or username is already registered" };
   294→      }
   295→      if (code === "auth/weak-password") {
   296→        return { success: false, error: "Password is too weak (min 6 chars)" };
   297→      }
   298→      if (code === "auth/invalid-email") {
   299→        return { success: false, error: "Invalid email address" };
   300→      }
   301→      return { success: false, error: "Signup failed. Please try again." };
   302→    }
   303→  },
   304→
   305→  login: async (username, password) => {
   306→    if (!auth || !db) {
   307→      return { success: false, error: "Firebase is not configured. See .env.local" };
   308→    }
   309→    const normalized = username.trim().toLowerCase();
   310→    try {
   311→      // Look up the username doc to check if user exists
   312→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   313→      if (!usernameDoc.exists()) {
   314→        return { success: false, error: "User not found. Please sign up first." };
   315→      }
   316→
   317→      // Use authEmail for login (the email used during Firebase Auth signup)
   318→      // NEVER use recoveryEmail for login — it’s only for password reset
   319→      const data = usernameDoc.data();
   320→      let loginEmail = data.authEmail || usernameToEmail(normalized);
   321→
   322→      // Auto-migrate old docs: if authEmail field doesn’t exist, set it now
   323→      if (!data.authEmail) {
   324→        const correctAuthEmail = usernameToEmail(normalized);
   325→        updateDoc(doc(db, "usernames", normalized), { authEmail: correctAuthEmail }).catch(() => {});
   326→        loginEmail = correctAuthEmail;
   327→      }
   328→
   329→      await signInWithEmailAndPassword(auth, loginEmail, password);
   330→      return { success: true };
   331→    } catch (err: unknown) {
   332→      const code = (err as { code?: string })?.code || "";
   333→      if (code === "auth/user-not-found") {
   334→        // authEmail in Firestore might not match Firebase Auth yet (pending email verification)
   335→        // Fall back to @et.app email
   336→        const fallbackEmail = usernameToEmail(normalized);
   337→        try {
   338→          await signInWithEmailAndPassword(auth, fallbackEmail, password);
   339→          // Login succeeded with fallback — update authEmail in Firestore to match
   340→          updateDoc(doc(db, "usernames", normalized), { authEmail: fallbackEmail }).catch(() => {});
   341→          return { success: true };
   342→        } catch {
   343→          // Fallback also failed — clean up orphaned Firestore data
   344→          try {
   345→            const data = (await getDoc(doc(db, "usernames", normalized))).data();
   346→            const uid = data?.uid;
   347→            if (uid) {
   348→              const expensesSnapshot = await getDocs(collection(db, "users", uid, "expenses"));
   349→              if (!expensesSnapshot.empty) {
   350→                const batch = writeBatch(db);
   351→                expensesSnapshot.docs.forEach((d) => batch.delete(d.ref));
   352→                await batch.commit();
   353→              }
   354→              await deleteDoc(doc(db, "users", uid, "settings", "config")).catch(() => {});
   355→            }
   356→          } catch { /* best effort cleanup */ }
   357→          await deleteDoc(doc(db, "usernames", normalized)).catch(() => {});
   358→          return { success: false, error: "User not found. Please sign up first." };
   359→        }
   360→      }
   361→      if (code === "auth/invalid-credential" || code === "auth/wrong-password") {
   362→        return { success: false, error: "Incorrect password" };
   363→      }
   364→      return { success: false, error: "Login failed. Please try again." };
   365→    }
   366→  },
   367→
   368→  logout: async () => {
   369→    if (!auth) return;
   370→    try {
   371→      await signOut(auth);
   372→    } catch {
   373→      // ignore sign out errors
   374→    }
   375→  },

   377→  lock: () => {
   378→    set({ isLocked: true });
   379→  },
   380→
   381→  unlock: async (password) => {
   382→    if (!auth || !auth.currentUser) return false;
   383→    try {
   384→      // Use the actual email from Firebase Auth (works for both old and new users)
   385→      const email = auth.currentUser.email!;
   386→      const credential = EmailAuthProvider.credential(email, password);
   387→      await reauthenticateWithCredential(auth.currentUser, credential);
   388→      set({ isLocked: false });
   389→      return true;
   390→    } catch {
   391→      return false;
   392→    }
   393→  },
   394→
   395→  deleteAccount: async (password) => {
   396→    if (!auth || !db || !auth.currentUser) {
   397→      return { success: false, error: "Not authenticated" };
   398→    }
   399→    const { uid, currentUser } = get();
   400→    if (!uid) return { success: false, error: "Not authenticated" };
   401→    try {
   402→      // Re-authenticate (use actual email from Firebase Auth)
   403→      const email = auth.currentUser.email!;
   404→      const credential = EmailAuthProvider.credential(email, password);
   405→      await reauthenticateWithCredential(auth.currentUser, credential);
   406→
   407→      // Delete all expenses from Firestore
   408→      const expensesRef = collection(db, "users", uid, "expenses");
   409→      const snapshot = await getDocs(expensesRef);
   410→      if (!snapshot.empty) {
   411→        const batch = writeBatch(db);
   412→        snapshot.docs.forEach((d) => batch.delete(d.ref));
   413→        await batch.commit();
   414→      }
   415→
   416→      // Delete settings document
   417→      try {
   418→        await deleteDoc(doc(db, "users", uid, "settings", "config"));
   419→      } catch {
   420→        // settings doc might not exist
   421→      }
   422→
   423→      // Delete username mapping
   424→      if (currentUser) {
   425→        try {
   426→          await deleteDoc(doc(db, "usernames", currentUser));
   427→        } catch {
   428→          // ignore
   429→        }
   430→      }
   431→
   432→      // Delete the Firebase Auth user
   433→      await firebaseDeleteUser(auth.currentUser);
   434→      return { success: true };
   435→    } catch (err: unknown) {
   436→      const code = (err as { code?: string })?.code || "";
   437→      if (code === "auth/wrong-password" || code === "auth/invalid-credential") {
   438→        return { success: false, error: "Incorrect password" };
   439→      }
   440→      return { success: false, error: "Failed to delete account. Please try again." };
   441→    }
   442→  },
   443→
   444→  forgotPassword: async (username) => {
   445→    if (!auth || !db) {
   446→      return { success: false, error: "Firebase not configured" };
   447→    }
   448→    const normalized = username.trim().toLowerCase();
   449→    try {
   450→      // Look up the user’s recovery email from Firestore
   451→      const usernameDoc = await getDoc(doc(db, "usernames", normalized));
   452→      if (!usernameDoc.exists()) {
   453→        return { success: false, error: "Username not found" };
   454→      }
   455→      const data = usernameDoc.data();
   456→      // Use recoveryEmail field (new), fall back to email field (old docs)
   457→      const recoveryEmail = data.recoveryEmail || data.email;
   458→
   459→      // Check if the user has a real recovery email
   460→      if (!recoveryEmail || recoveryEmail.endsWith("@et.app")) {
   461→        return {
   462→          success: false,
   463→          error: "This account was created without a recovery email. Please create a new account with your email to enable password reset.",
   464→        };
   465→      }
   466→
   467→      // Send Firebase password reset email
   468→      await sendPasswordResetEmail(auth, recoveryEmail, {
   469→        url: "https://expense-tracker-five-alpha-69.vercel.app/",
   470→      });
   471→      return { success: true, email: recoveryEmail };
   472→    } catch {
   473→      return { success: false, error: "Failed to send reset email. Please try again." };
   474→    }
   475→  },
   476→
   477→  updateUserEmail: async (newEmail, password) => {
   478→    if (!auth?.currentUser) {
   479→      return { success: false, error: "Not authenticated" };
   480→    }
   481→    const { currentUser } = get();
   482→    if (!currentUser || !db) {
   483→      return { success: false, error: "Not authenticated" };
   484→    }
   485→    const trimmedEmail = newEmail.trim().toLowerCase();
   486→    if (!trimmedEmail || !trimmedEmail.includes("@") || !trimmedEmail.includes(".")) {
   487→      return { success: false, error: "Please enter a valid email address" };
   488→    }
   489→    if (!password || password.length < 6) {
   490→      return { success: false, error: "Please enter your password to confirm" };
   491→    }
   492→
   493→    // Check if this email is already used by another account
   494→    try {
   495→      const q = query(
   496→        collection(db, "usernames"),
   497→        where("recoveryEmail", "==", trimmedEmail)
   498→      );
   499→      const snapshot = await getDocs(q);
   500→      for (const d of snapshot.docs) {
   501→        if (d.id !== currentUser) {
   502→          return { success: false, error: "This email is already used by another account" };
   503→        }
   504→      }
   505→      // Also check old field name for backward compat
   506→      const q2 = query(
   507→        collection(db, "usernames"),
   508→        where("email", "==", trimmedEmail)
   509→      );
   510→      const snapshot2 = await getDocs(q2);
   511→      for (const d of snapshot2.docs) {
   512→        if (d.id !== currentUser) {
   513→          // Only flag if this doc doesn't already have recoveryEmail set
   514→          if (!d.data().recoveryEmail) {
   515→            return { success: false, error: "This email is already used by another account" };
   516→          }
   517→        }
   518→      }
   519→    } catch (err) {
   520→      // If query fails, try fallback: fetch all docs and check manually
   521→      try {
   522→        const allDocs = await getDocs(collection(db, "usernames"));
   523→        for (const d of allDocs.docs) {
   524→          if (d.id === currentUser) continue;
   525→          const data = d.data();
   526→          const em = data.recoveryEmail || data.email;
   527→          if (em && em.toLowerCase() === trimmedEmail) {
   528→            return { success: false, error: "This email is already used by another account" };
   529→          }
   530→        }
   531→      } catch {
   532→        // If even the fallback fails, allow the save (security rules may block reads)
   533→      }
   534→    }
   535→
   536→    // Step 1: Re-authenticate, then send verification email to the new address
   537→    try {
   538→      // Capture the current user reference before re-auth (it may change after re-auth)
   539→      const userBefore = auth.currentUser!;
   540→      const currentAuthEmail = userBefore.email!;
   541→      const credential = EmailAuthProvider.credential(currentAuthEmail, password);
   542→      const result = await reauthenticateWithCredential(userBefore, credential);
   543→      // Use the refreshed user reference from re-auth result
   544→      const refreshedUser = result.user;
   545→      if (refreshedUser.email !== trimmedEmail) {
   546→        // Firebase project requires email verification before changing — use verifyBeforeUpdateEmail
   547→        // This sends a verification link to the new email; the email changes only after the user clicks it
   548→        await firebaseVerifyBeforeUpdateEmail(refreshedUser, trimmedEmail);
   549→      }
   550→    } catch (authErr: unknown) {
   551→      const code = (authErr as { code?: string })?.code || "";
   552→      const message = (authErr as { message?: string })?.message || "";
   553→      console.error("Firebase Auth email verification failed:", code, message);
   554→      if (code === "auth/wrong-password" || code === "auth/invalid-credential") {
   555→        return { success: false, error: "Incorrect password. Please try again." };
   556→      }
   557→      if (code === "auth/email-already-in-use") {
   558→        return { success: false, error: "This email is already used by another Firebase account" };
   559→      }
   560→      if (code === "auth/too-many-requests") {
   561→        return { success: false, error: "Too many attempts. Please wait a moment and try again." };
   562→      }
   563→      if (code === "auth/invalid-email") {
   564→        return { success: false, error: "Invalid email format. Please check and try again." };
   565→      }
   566→      return { success: false, error: `Failed to send verification: ${message || "Unknown error"}` };
   567→    }
   568→
   569→    // Step 2: Save recovery email to Firestore immediately (so UI shows it)
   570→    try {
   571→      await updateDoc(doc(db, "usernames", currentUser), {
   572→        recoveryEmail: trimmedEmail,
   573→        email: trimmedEmail,
   574→        authEmail: trimmedEmail,
   575→      });
   576→    } catch {
   577→      // Firestore update failed but verification email was sent — acceptable
   578→    }
   579→
   580→    // Update local state
   581→    set({ userEmail: trimmedEmail, needsEmailUpdate: false });
   582→    return { success: true, verificationSent: true };
   583→  },
   584→}));

   673→// ============================================================
   674→// SETTINGS STORE — backed by Cloud Firestore
   675→// ============================================================
   676→
   677→interface SettingsState {
   678→  currencyCode: string;
   679→  monthlyBudget: number; // legacy single budget (kept for migration)
   680→  monthlyBudgets: Record<string, number>; // per-month budgets: "YYYY-MM" -> amount
   681→  isLoading: boolean;
   682→  _unsubscribe: Unsubscribe | null;
   683→
   684→  subscribeToSettings: (uid: string) => void;
   685→  unsubscribeFromSettings: () => void;
   686→  saveSettings: (currencyCode: string) => Promise<void>;
   687→  saveBudgetForMonth: (monthKey: string, amount: number) => Promise<void>;
   688→  getBudgetForMonth: (monthKey: string) => number;
   689→}
   690→
   691→export const useSettingsStore = create<SettingsState>()((set, get) => ({
   692→  currencyCode: "BHD",
   693→  monthlyBudget: 0,
   694→  monthlyBudgets: {},
   695→  isLoading: false,
   696→  _unsubscribe: null,
   697→
   698→  subscribeToSettings: (uid: string) => {
   699→    const prev = get()._unsubscribe;
   700→    if (prev) prev();
   701→
   702→    if (!db) return;
   703→
   704→    set({ isLoading: true });
   705→    const docRef = doc(db, "users", uid, "settings", "config");
   706→    const unsub = onSnapshot(
   707→      docRef,
   708→      (snapshot) => {
   709→        if (snapshot.exists()) {
   710→          const data = snapshot.data();
   711→          const budgets = data.monthlyBudgets || {};
   712→          const legacyBudget = data.monthlyBudget || 0;
   713→
   714→          // Migrate old single monthlyBudget to current month if no per-month budgets exist
   715→          if (legacyBudget > 0 && Object.keys(budgets).length === 0) {
   716→            const currentMonth = `${new Date().getFullYear()}-${String(new Date().getMonth() + 1).padStart(2, "0")}`;
   717→            budgets[currentMonth] = legacyBudget;
   718→            // Write migrated data back to Firestore
   719→            setDoc(docRef, { monthlyBudgets: budgets, monthlyBudget: 0 }, { merge: true }).catch(() => {});
   720→          }
   721→
   722→          set({
   723→            currencyCode: data.currencyCode || "BHD",
   724→            monthlyBudget: legacyBudget,
   725→            monthlyBudgets: budgets,
   726→            isLoading: false,
   727→          });
   728→        } else {
   729→          set({ isLoading: false });
   730→        }
   731→      },
   732→      () => {
   733→        set({ isLoading: false });
   734→      }
   735→    );
   736→
   737→    set({ _unsubscribe: unsub });
   738→  },
   739→
   740→  unsubscribeFromSettings: () => {
   741→    const prev = get()._unsubscribe;
   742→    if (prev) {
   743→      prev();
   744→      set({ _unsubscribe: null });
   745→    }
   746→  },
   747→
   748→  saveSettings: async (currencyCode) => {
   749→    const { uid } = useAuthStore.getState();
   750→    if (!db || !uid) return;
   751→    const docRef = doc(db, "users", uid, "settings", "config");
   752→    await setDoc(docRef, { currencyCode }, { merge: true });
   753→  },
   754→
   755→  saveBudgetForMonth: async (monthKey, amount) => {
   756→    const { uid } = useAuthStore.getState();
   757→    if (!db || !uid) return;
   758→    const docRef = doc(db, "users", uid, "settings", "config");
   759→
   760→    // Optimistic update — update the store immediately so UI reflects instantly
   761→    const currentBudgets = get().monthlyBudgets;
   762→    const optimistic = { ...currentBudgets };
   763→    if (amount > 0) {
   764→      optimistic[monthKey] = amount;
   765→    } else {
   766→      delete optimistic[monthKey];
   767→    }
   768→    set({ monthlyBudgets: optimistic });
   769→
   770→    // Persist to Firestore using dot notation (only updates the specific month key)
   771→    try {
   772→      if (amount > 0) {
   773→        await updateDoc(docRef, { ["monthlyBudgets." + monthKey]: amount });
   774→      } else {
   775→        await updateDoc(docRef, { ["monthlyBudgets." + monthKey]: firebaseDeleteField() });
   776→      }
   777→    } catch {
   778→      // onSnapshot will correct the store if the write fails
   779→    }
   780→  },
   781→
   782→  getBudgetForMonth: (monthKey) => {
   783→    const { monthlyBudgets } = get();
   784→    return monthlyBudgets[monthKey] || 0;
   785→  },
   786→}));
